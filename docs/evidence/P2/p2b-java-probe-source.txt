import android.opengl.EGL14;
import android.opengl.EGLConfig;
import android.opengl.EGLContext;
import android.opengl.EGLDisplay;
import android.opengl.EGLSurface;
import android.opengl.GLES20;
import java.util.Arrays;
import java.nio.ByteBuffer;
import java.nio.ByteOrder;
import java.nio.FloatBuffer;

public class P2GlInfo {
    static int shader(int type, String source) {
        int id = GLES20.glCreateShader(type);
        GLES20.glShaderSource(id, source);
        GLES20.glCompileShader(id);
        int[] ok = new int[1];
        GLES20.glGetShaderiv(id, GLES20.GL_COMPILE_STATUS, ok, 0);
        if (ok[0] == 0) throw new RuntimeException(GLES20.glGetShaderInfoLog(id));
        return id;
    }
    static void astcProbe() {
        int program = GLES20.glCreateProgram();
        GLES20.glAttachShader(program, shader(GLES20.GL_VERTEX_SHADER,
            "attribute vec2 p; varying vec2 uv; void main(){uv=p*0.5+0.5;gl_Position=vec4(p,0.0,1.0);}"));
        GLES20.glAttachShader(program, shader(GLES20.GL_FRAGMENT_SHADER,
            "precision mediump float; varying vec2 uv; uniform sampler2D t; void main(){gl_FragColor=texture2D(t,uv);}"));
        GLES20.glLinkProgram(program);
        GLES20.glUseProgram(program);
        int[] tex = new int[1];
        GLES20.glGenTextures(1, tex, 0);
        GLES20.glBindTexture(GLES20.GL_TEXTURE_2D, tex[0]);
        GLES20.glTexParameteri(GLES20.GL_TEXTURE_2D, GLES20.GL_TEXTURE_MIN_FILTER, GLES20.GL_NEAREST);
        GLES20.glTexParameteri(GLES20.GL_TEXTURE_2D, GLES20.GL_TEXTURE_MAG_FILTER, GLES20.GL_NEAREST);
        ByteBuffer block = ByteBuffer.allocateDirect(16).order(ByteOrder.LITTLE_ENDIAN);
        block.putLong(0xfffffffffffffdfcL).putShort((short)65535).putShort((short)0).putShort((short)0).putShort((short)65535).rewind();
        GLES20.glCompressedTexImage2D(GLES20.GL_TEXTURE_2D, 0, 0x93b0, 4, 4, 0, 16, block);
        System.out.println("ASTC_UPLOAD_ERROR=" + GLES20.glGetError());
        FloatBuffer vertices = ByteBuffer.allocateDirect(32).order(ByteOrder.nativeOrder()).asFloatBuffer();
        vertices.put(new float[]{-1,-1,1,-1,-1,1,1,1}).rewind();
        int loc = GLES20.glGetAttribLocation(program, "p");
        GLES20.glVertexAttribPointer(loc, 2, GLES20.GL_FLOAT, false, 0, vertices);
        GLES20.glEnableVertexAttribArray(loc);
        GLES20.glUniform1i(GLES20.glGetUniformLocation(program, "t"), 0);
        GLES20.glViewport(0, 0, 16, 16);
        GLES20.glClearColor(0, 1, 0, 1);
        GLES20.glClear(GLES20.GL_COLOR_BUFFER_BIT);
        GLES20.glDrawArrays(GLES20.GL_TRIANGLE_STRIP, 0, 4);
        ByteBuffer pixel = ByteBuffer.allocateDirect(4);
        GLES20.glReadPixels(8, 8, 1, 1, GLES20.GL_RGBA, GLES20.GL_UNSIGNED_BYTE, pixel);
        System.out.printf("ASTC_RED_PIXEL=%d,%d,%d,%d GL_ERROR=%d%n", pixel.get(0)&255, pixel.get(1)&255, pixel.get(2)&255, pixel.get(3)&255, GLES20.glGetError());
        GLES20.glDeleteTextures(1, tex, 0);
        GLES20.glDeleteProgram(program);
    }
    public static void main(String[] args) {
        EGLDisplay display = EGL14.eglGetDisplay(EGL14.EGL_DEFAULT_DISPLAY);
        int[] version = new int[2];
        if (!EGL14.eglInitialize(display, version, 0, version, 1)) throw new RuntimeException("eglInitialize " + EGL14.eglGetError());
        System.out.println("EGL_VERSION=" + EGL14.eglQueryString(display, EGL14.EGL_VERSION));
        System.out.println("EGL_EXTENSIONS=" + EGL14.eglQueryString(display, EGL14.EGL_EXTENSIONS));
        EGLConfig[] configs = new EGLConfig[1];
        int[] count = new int[1];
        int[] attrs = {EGL14.EGL_SURFACE_TYPE, EGL14.EGL_PBUFFER_BIT, EGL14.EGL_RENDERABLE_TYPE, EGL14.EGL_OPENGL_ES2_BIT, EGL14.EGL_RED_SIZE, 8, EGL14.EGL_GREEN_SIZE, 8, EGL14.EGL_BLUE_SIZE, 8, EGL14.EGL_NONE};
        if (!EGL14.eglChooseConfig(display, attrs, 0, configs, 0, 1, count, 0) || count[0] == 0) throw new RuntimeException("eglChooseConfig");
        EGLContext context = EGL14.eglCreateContext(display, configs[0], EGL14.EGL_NO_CONTEXT, new int[]{EGL14.EGL_CONTEXT_CLIENT_VERSION, 3, EGL14.EGL_NONE}, 0);
        EGLSurface surface = EGL14.eglCreatePbufferSurface(display, configs[0], new int[]{EGL14.EGL_WIDTH, 16, EGL14.EGL_HEIGHT, 16, EGL14.EGL_NONE}, 0);
        if (!EGL14.eglMakeCurrent(display, surface, surface, context)) throw new RuntimeException("eglMakeCurrent " + EGL14.eglGetError());
        System.out.println("GL_VENDOR=" + GLES20.glGetString(GLES20.GL_VENDOR));
        System.out.println("GL_RENDERER=" + GLES20.glGetString(GLES20.GL_RENDERER));
        System.out.println("GL_VERSION=" + GLES20.glGetString(GLES20.GL_VERSION));
        String[] extensions = GLES20.glGetString(GLES20.GL_EXTENSIONS).split(" ");
        Arrays.sort(extensions);
        for (String ext : extensions) System.out.println("EXT " + ext);
        GLES20.glGetIntegerv(GLES20.GL_NUM_COMPRESSED_TEXTURE_FORMATS, count, 0);
        int[] formats = new int[count[0]];
        GLES20.glGetIntegerv(GLES20.GL_COMPRESSED_TEXTURE_FORMATS, formats, 0);
        for (int format : formats) System.out.printf("COMPRESSED_FORMAT=0x%04x%n", format);
        System.out.println("GL_ERROR=" + GLES20.glGetError());
        astcProbe();
        EGL14.eglMakeCurrent(display, EGL14.EGL_NO_SURFACE, EGL14.EGL_NO_SURFACE, EGL14.EGL_NO_CONTEXT);
        EGL14.eglDestroySurface(display, surface);
        EGL14.eglDestroyContext(display, context);
        EGL14.eglTerminate(display);
    }
}
